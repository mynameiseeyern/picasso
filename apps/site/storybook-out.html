Typography

/* my-brand — generated by the Picasso theme builder. Do not hand-edit; regenerate. */
/* Paste ABOVE the axis blocks in packages/tokens/themes.css. */
[data-brand="my-brand"] {
  /* brand ramp, OKLCH from #007FAB; 600–950 solved for white text */
  --palette-brand-50: #EFF8FD;
  --palette-brand-100: #D9EEF9;
  --palette-brand-200: #B7E0F6;
  --palette-brand-300: #8ECDEE;
  --palette-brand-400: #5FB5DE;
  --palette-brand-500: #3299C6;
  --palette-brand-600: #007DA8;
  --palette-brand-700: #006386;
  --palette-brand-800: #004D6A;
  --palette-brand-900: #003B52;
  --palette-brand-950: #002838;

  /* the brand's defaults for the axes */
  --radius: 0.625rem;
  --space-1: 0.25rem;
  --type-family-sans: "Inter", ui-sans-serif, system-ui, -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif;
  --type-family-mono: "JetBrains Mono", ui-monospace, SFMono-Regular, Menlo, Monaco, Consolas, "Liberation Mono", monospace;
}

/* on-colours, measured — white where it passes 4.5:1, dark ink where it cannot */
[data-brand="my-brand"] {
  --fg-on-primary-emphasis: var(--palette-white); /* on #007DA8: white 4.67, dark 3.87 */
}
.dark [data-brand="my-brand"], [data-brand="my-brand"].dark {
  --fg-on-primary-emphasis: var(--palette-neutral-950); /* on #3299C6: white 3.23, dark 5.6 */
}
